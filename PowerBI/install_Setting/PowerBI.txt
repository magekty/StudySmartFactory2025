from sklearn.datasets import load_iris
import pandas as pd
import numpy as np
iris = load_iris()
df = pd.DataFrame(iris['data'], columns=iris['feature_names'])
df['target'] = iris['target']
from sklearn.tree import DecisionTreeClassifier
X = iris['data']
y = iris['target']
clf = DecisionTreeClassifier()
clf.fit(X, y) 
y = clf.predict(X)
df['prediction'] = y
